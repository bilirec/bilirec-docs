以下是一份完整的設定示例，可直接複製到 `.env` 或 `.env.local`（`KEY=VALUE` 格式，不要加 `export`）：

```bash
BILIBILI_LOGIN_MODE=controller
# HOST=0.0.0.0
PORT=8080
# SERVER_CRT=/path/to/server.crt
# SERVER_KEY=/path/to/server.key
# 監控指標（預設關閉，啟用後由 VictoriaMetrics / Grafana 抓取）
# METRICS_ENABLED=true
# METRICS_HOST=          # 留空監聽所有網卡，便於跨機器抓取
# METRICS_PORT=9090
# VictoriaLogs 遠端日誌（預設關閉）
# VICTORIALOGS_ENABLED=true
# VICTORIALOGS_URL=http://127.0.0.1:9428
# VICTORIALOGS_STREAM_FIELDS=app,logger
# 本地日誌檔案（預設關閉；隊列滿時預設 drop，不阻塞錄製，詳見 local-logs）
# LOCAL_LOGS_ENABLED=true
# LOCAL_LOGS_PATH=./logs/bilirec.log
# LOCAL_LOGS_FORMAT=pretty
# LOCAL_LOGS_OVERFLOW=drop
FRP_ENABLED=true
FRP_SERVER=tunnel.bilirec.org:7000
FRP_BASE_DOMAIN=tunnel.bilirec.org
TRUSTED_PROXIES=161.33.159.26
FRP_HTTPS=false
FRP_SCHEME_HTTPS=true
MAX_CONCURRENT_RECORDINGS=3
MAX_RECORDING_HOURS=5
MAX_RECOVERY_ATTEMPTS=15
MAX_RETRY_MINUTES=10
RECORDING_RECOVERY_DURATION=preserve
# SUBCHECK_ROOMS_PER_SHARD=50
# SUBCHECK_TICK_SECS=10
# SUBCHECK_MIN_INTERVAL_SECS=60
# SUBCHECK_MAX_INTERVAL_SECS=300
# SUBCHECK_MAX_SHARDS=32
# SUBCHECK_JITTER_SECS=2  # 打散輪詢節奏，降低反爬風控
OUTPUT_DIR=/path/to/records
SECRET_DIR=/path/to/secrets
NO_CONVERT_IF_INVALID=false
DATABASE_DIR=/path/to/database
CONVERT_TO_MP4=false
DELETE_SOURCE_AFTER_CONVERT=false
CLOUDCONVERT_THRESHOLD=1073741824
CLOUDCONVERT_API_KEY=
CLOUDCONVERT_CHECK_INTERVAL_SECS=180
CLOUDCONVERT_MAX_CONCURRENT_DOWNLOADS=1
CLOUDCONVERT_ALLOW_DURING_RECORDING=false
CLOUDCONVERT_ALLOW_DURING_RECORDING_MAX_ACTIVE_RECORDINGS=1
FFMPEG_CHECK_INTERVAL_SECS=60
FFMPEG_MAX_CONCURRENT_TASKS=1
FFMPEG_ALLOW_DURING_RECORDING=false
FFMPEG_ALLOW_DURING_RECORDING_MAX_ACTIVE_RECORDINGS=1
PUBLIC_BASE_URL=http://localhost:8080
FRONTEND_URL=https://app.bilirec.org
WEBPUSH_SUBSCRIBER=mailto:webpush@example.com
UPLOAD_BUFFER_SIZE=5242880
DOWNLOAD_BUFFER_SIZE=5242880
STREAM_WRITER_BUFFER_SIZE=1048576
READ_STREAM_BYTES_POOL_SIZE=524288
READ_STREAM_CHAN_BUFFER_SIZE=16
READ_STREAM_BYTES_POOL_SIZE_HIGH=1048576
READ_STREAM_CHAN_BUFFER_SIZE_HIGH=48
LIVE_STREAM_WRITER_BUFFER_SIZE=8388608
LIVE_STREAM_WRITER_SYNC_PERIOD_SECS=0
LIVE_STREAM_WRITER_COLD_CACHE_RELEASE_SECS=60
LIVE_STREAM_WRITER_FLUSH_PERIOD_SECS=15
LIVE_STREAM_WRITER_CHAN_BUFFER_SIZE=64
LIVE_STREAM_WRITER_BYTES_POOL_SIZE=524288
LIVE_STREAM_WRITER_BYTES_POOL_SIZE_HIGH=1048576
SKIP_SMALL_FLUSH_THRESHOLD=1048576
SKIP_SMALL_FLUSH=true
SEQUENTIAL_WRITE=true
# 預設已開 DROP + 冷釋放；SSD/HDD 高性能場景可關閉：
# DROP_FILE_PAGE_CACHE=false
# LIVE_STREAM_WRITER_COLD_CACHE_RELEASE_SECS=0
# 彈幕錄製（是否錄製由房間設定或 record_danmaku 控制）
DANMAKU_OUTPUT_FORMAT=jsonl
DANMAKU_OVERFLOW_POLICY=drop
DANMAKU_WRITER_BUFFER_SIZE=262144
DANMAKU_WRITER_MIN_PERIODIC_FLUSH_BYTES=16384
DANMAKU_WRITER_FLUSH_PERIOD_SECS=15
DANMAKU_WRITER_SYNC_PERIOD_SECS=0
DANMAKU_CHAN_BUFFER_SIZE=256
DANMAKU_BYTES_POOL_SIZE=4096
MIN_DISK_SPACE_BYTES=5368709120
JWT_SECRET=bilirec_secret
DEBUG=false
USERNAME=admin
PASSWORD=changeme
# 本示例啟用了官方 FRP，並通過 app.bilirec.org 遠端管理，須開啟生產模式
PRODUCTION_MODE=true
```

各變數的詳細說明見[環境變數參考](/zh-tw/configuration/env-reference/)。針對不同硬件的調優方案見 [樹莓派預設](/zh-tw/configuration/pi5-defaults/)、[高碼率 / 4K 調參](/zh-tw/configuration/high-bitrate-tuning/)、[SSD 調優](/zh-tw/configuration/ssd-tuning/) 與 [HDD/NAS 調優](/zh-tw/configuration/hdd-nas-tuning/)。
